function nav(id, el) {
  document.querySelectorAll('.pg').forEach(p => p.classList.remove('on'));
  var p = document.getElementById('pg-' + id);
  if (p) p.classList.add('on');
  document.querySelectorAll('.ni').forEach(n => n.classList.remove('on'));
  if (el) el.classList.add('on');
}
function tab(pgId, tcId, clickEl) {
  var pg = document.getElementById('pg-' + pgId);
  if (!pg) return;
  pg.querySelectorAll('.tc').forEach(t => t.classList.remove('on'));
  var tc = document.getElementById(tcId);
  if (tc) tc.classList.add('on');
  if (clickEl) {
    var bar = clickEl.closest ? clickEl.closest('.tabrow') : null;
    if (bar) { bar.querySelectorAll('.t').forEach(t => t.classList.remove('on')); clickEl.classList.add('on'); }
  } else if (tc) {
    var all = Array.from(pg.querySelectorAll('.tc'));
    var idx = all.indexOf(tc);
    var tbars = pg.querySelectorAll('.tabrow .t');
    if (idx >= 0 && tbars[idx]) { tbars.forEach(t => t.classList.remove('on')); tbars[idx].classList.add('on'); }
  }
}
function setSbt(el) {
  var b = el.closest ? el.closest('.stbar') : null;
  if (b) { b.querySelectorAll('.sbt').forEach(s => s.classList.remove('on')); el.classList.add('on'); }
}
function openWsMod()  { document.getElementById('wsMod').classList.add('on'); }
function closeWsMod() { document.getElementById('wsMod').classList.remove('on'); }
function toggleNewWs() {
  var f = document.getElementById('newWsForm');
  f.style.display = f.style.display === 'none' ? 'block' : 'none';
}
function testConn() {
  var d = document.getElementById('csD'), m = document.getElementById('csM');
  d.className = 'csd tst'; m.textContent = 'Testing connection…';
  setTimeout(function(){ d.className='csd ok'; m.textContent='✓ Connected · Unity Catalog OK · regression_catalog found'; }, 1800);
}
function startProgress() {
  var fill = document.getElementById('progFill');
  var pct  = document.getElementById('progPct');
  if (!fill) return;
  var w = 62;
  var iv = setInterval(function() {
    w = Math.min(w + 1, 97);
    fill.style.width = w + '%';
    if (pct) pct.textContent = '~' + w + '% · est. ' + Math.max(0, Math.round((97 - w) * 0.6)) + 's remaining';
    if (w >= 97) clearInterval(iv);
  }, 350);
}
