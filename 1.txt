            document.querySelectorAll('.view-section').forEach(v => v.classList.remove('active'));

            document.getElementById('config-modal').classList.remove('hidden');

        function closeConfig() {
            document.getElementById('config-modal').classList.add('hidden');

        function addLog(msg, type = 'info') {
            const time = new Date().toLocaleTimeString();
            const logEl = document.createElement('div');
            logEl.className = "flex gap-3 animate-in fade-in slide-in-from-left-2 duration-300";
            logEl.innerHTML = `
                <span class="text-slate-600 shrink-0">[${time}]</span>
                <span class="${type === 'success' ? 'text-green-500' : 'text-primary'} font-bold shrink-0">[${type.toUpperCase()}]</span>

            `;
            const container = document.getElementById('log-container');
            container.insertBefore(logEl, container.firstChild);
        }

        function updateStep(id, status, metrics) {
            const card = document.getElementById(`step-${id}`);
            card.classList.remove('opacity-40', 'active', 'completed');
            if (status === 'running') {
                card.classList.add('active');
                card.querySelector('.status-icon').innerHTML = '<svg class="animate-spin text-primary" width="20" height="20" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2"><path d="M21 12a9 9 0 1 1-6.219-8.56"/></svg>';

                card.classList.add('completed');

            }
            if (metrics) card.querySelector('.step-metrics').innerText = metrics;
        }
